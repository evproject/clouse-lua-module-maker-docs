API shape

Lua entry points

on_load(ctx)

Initialize state, load saved JSON, and print setup messages.

on_tick(ctx)

Run client-tick logic and read player or world snapshots.

on_render_hud(ctx, draw)

Draw simple HUD overlays from the HUD template.

on_command(ctx, args)

Build command-style modules with subcommand parsing.

local module = {
  id = "quick_start",
  name = "Quick Start"
}

function module.on_tick(ctx)
  if not ctx.setting("enabled", true) then
    return
  end

  local player = ctx.player()
  ctx.log("tick " .. player.name)
end

return module
